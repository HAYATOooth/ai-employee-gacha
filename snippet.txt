function rarityClass(c){return c.rarity==='UR'?'biz-ur':c.rarity==='SSR'?'biz-ssr':c.rarity==='SR'?'biz-sr':'biz-r'}
function rarityColor(c){return c.rarity==='UR'?'#ffd45c':c.rarity==='SSR'?'#b58cff':c.rarity==='SR'?'#62baff':'#d8e2f5'}
function startCinema(num){
 initAudio();tone(120,.22,'triangle',.045);playBgm();cinemaMode=num;cinemaResult=null;cinemaStart=performance.now();clearCinemaTimers();
 const o=document.getElementById('cinema');o.className='cinema show';o.classList.remove('biz-ur','biz-ssr','biz-sr','biz-r','scene-purple','scene-gold','scene-ur');setScene('city');resetGachaVisual();
 document.getElementById('cProgressBar').style.width='0%';document.getElementById('cTime').textContent='00.0 / 10.0';
 if(num===10)prepareCandidates();else cinemaResult=weighted();
 requestAnimationFrame(()=>{document.getElementById('gachaOrb').classList.add('appear');});
 cinemaTimer=setInterval(updateCinemaClock,50);
 cinemaTimeouts.push(setTimeout(()=>gachaStage(1),1800));
 cinemaTimeouts.push(setTimeout(()=>gachaStage(2),3600));
 cinemaTimeouts.push(setTimeout(()=>gachaStage(3),5200));
 cinemaTimeouts.push(setTimeout(()=>gachaStage(4),6800));
 cinemaTimeouts.push(setTimeout(()=>gachaStage(5),7900));
 cinemaTimeouts.push(setTimeout(()=>finishCinema(false),10000));
}
function updateCinemaClock(){const elapsed=Math.min(10000,performance.now()-cinemaStart);document.getElementById('cProgressBar').style.width=(elapsed/10000*100)+'%';document.getElementById('cTime').textContent=(elapsed/1000).toFixed(1)+' / 10.0'}
function gachaStage(stage){
 const orb=document.getElementById('gachaOrb'),rar=document.getElementById('gachaRarity'),img=document.getElementById('gachaResultImg'),name=document.getElementById('gachaName'),sub=document.getElementById('gachaSub'),burst=document.getElementById('gachaBurst'),rays=document.getElementById('gachaRays');

 }
